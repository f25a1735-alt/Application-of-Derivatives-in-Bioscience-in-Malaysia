    <script>
        // Generate realistic data points over a 400-day feedlot/growth cycle
        const labels = [];
        const weightData = [];
        const rateData = [];

        // f(x) = -0.000002x³ + 0.0015x² + 0.15x + 25
        function getWeight(x) {
            return (-0.000002 * Math.pow(x, 3)) + (0.0015 * Math.pow(x, 2)) + (0.15 * x) + 25;
        }

        // f'(x) = -0.000006x² + 0.003x + 0.15
        function getRate(x) {
            return (-0.000006 * Math.pow(x, 2)) + (0.003 * x) + 0.15;
        }

        // Populate data points every 10 days
        for (let x = 0; x <= 400; x += 10) {
            labels.push(x);
            weightData.push(getWeight(x).toFixed(1));
            rateData.push(getRate(x).toFixed(3));
        }

        // Chart.js Setup
        const ctx = document.getElementById('growthChart').getContext('2d');
        const growthChart = new Chart(ctx, {
            type: 'line',
            data: {
                labels: labels,
                datasets: [
                    {
                        label: 'Cattle Weight (f(x) in kg)',
                        data: weightData,
                        borderColor: '#059669', // Emerald 600
                        backgroundColor: 'rgba(5, 150, 105, 0.1)',
                        yAxisID: 'yWeight',
                        borderWidth: 3,
                        tension: 0.2,
                        pointRadius: 3,
                        pointHoverRadius: 7
                    },
                    {
                        label: 'Growth Rate / ADG (f\'(x) in kg/day)',
                        data: rateData,
                        borderColor: '#06b6d4', // Cyan 500
                        borderDash: [5, 5],
                        yAxisID: 'yRate',
                        borderWidth: 2,
                        tension: 0.2,
                        pointRadius: 0 // Keep it clean, hide points for the derivative line
                    }
                ]
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                animation: {
                    duration: 2000,
                    easing: 'easeOutQuart'
                },
                scales: {
                    x: {
                        title: {
                            display: true,
                            text: 'Age of Animal (Days)',
                            font: { weight: 'bold' }
                        }
                    },
                    yWeight: {
                        type: 'linear',
                        position: 'left',
                        title: {
                            display: true,
                            text: 'Total Weight (kg)',
                            font: { weight: 'bold', color: '#059669' }
                        },
                        min: 0,
                        max: 200
                    },
                    yRate: {
                        type: 'linear',
                        position: 'right',
                        title: {
                            display: true,
                            text: 'Growth Rate (kg/Day)',
                            font: { weight: 'bold', color: '#06b6d4' }
                        },
                        min: 0,
                        max: 0.6,
                        grid: {
                            drawOnChartArea: false // prevent grid lines mixing
                        }
                    }
                },
                plugins: {
                    tooltip: {
                        mode: 'index',
                        intersect: false
                    }
                },
                // Handle click events to display derivative analysis
                onClick: (e, elements) => {
                    if (elements.length > 0) {
                        const index = elements[0].index;
                        const day = labels[index];
                        const weight = weightData[index];
                        const rate = rateData[index];
                        
                        updateDiagnostic(day, weight, rate);
                    }
                }
            }
        });

        // Function to handle the calculation and generate layman text on click
        function updateDiagnostic(day, weight, rate) {
            document.getElementById('defaultPrompt').classList.add('hidden');
            const content = document.getElementById('diagnosticContent');
            content.classList.remove('hidden');

            document.getElementById('outDay').innerText = day;
            document.getElementById('outWeight').innerText = weight;
            document.getElementById('outRate').innerText = rate;

            let explanation = "";

            // Dynamic logic simulating real animal husbandry strategies
            if (day <= 100) {
                explanation = `👶 Early Growth Phase (Weaning to Juvenile): The calf's growth rate is positive and increasing ($f'(x) = ${rate}\text{ kg/day}$). It is building bone and muscle framework. Keep baseline nutrient-dense feed steady.`;
            } else if (day > 100 && day <= 250) {
                explanation = `🚀 Peak Growth Acceleration! At day ${day}, the animal is highly efficient. The derivative is at or near its maximum ($${rate}\text{ kg/day}$). Action: This is your prime profit window! Allocate your highest quality silage and protein concentrates right now to maximize meat yield.`;
            } else if (day > 250 && day <= 350) {
                explanation = `⚖️ Growth Deceleration Phase: The total weight is still going up, but look at the derivative—it has dropped to $${rate}\text{ kg/day}$. The animal's biological growth is tapering off. Action: Transition away from expensive high-protein growth feeds to cheaper maintenance rations to preserve profit margins.`;
            } else {
                explanation = `🥩 Mature Market Phase: The derivative has dropped significantly ($${rate}\text{ kg/day}$), meaning the animal is barely gaining any more weight. Feeding it further will cost more than the meat is worth. Action: Schedule the animal for processing/market immediately.`;
            }

            document.getElementById('outExplain').innerHTML = explanation;
            
            // Give a little visual flash feedback to show change
            const box = document.getElementById('analysisBox');
            box.classList.add('border-cyan-400');
            setTimeout(() => box.classList.remove('border-cyan-400'), 400);
        }
    </script>
